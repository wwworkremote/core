def call
  SOURCES.each do |source|
    source_dir = File.join(BASE_DIR, source)
    next unless Dir.exist?(source_dir)

    Dir.glob(File.join(source_dir, '*.eml')).each do |file_path|
      process_file(file_path, source)
    end
  end
end
